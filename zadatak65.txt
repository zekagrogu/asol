            int x = listBox1.Items.Count;

        private void button4_Click(object sender, EventArgs e)
        {
            button2.PerformClick();
            preskoceno++;
            label6.Text = "Preskocena:" + preskoceno;
            if (progressBar1.Value < progressBar1.Maximum)
                progressBar1.Value += 1;
            else
            {
                MessageBox.Show("Kraj kviza sa osvojenih poena:"+poeni);
                timer1.Stop();
                Reset();
                return;
            }
        }

        private void button3_Click(object sender, EventArgs e)
        {
            int tacni = 0;
            bool greska = false;
            if (p1 == 1 && odgk == 1) tacni++;
            if (p1 == 1 && odgk == 0) greska = true;

            if (p2 == 1 && odgk1 == 1) tacni++;
            if (p2 == 1 && odgk1 == 0) greska = true;

            if (p3 == 1 && odgk2 == 1) tacni++;
            if (p3 == 1 && odgk2 == 0) greska = true;

            if (p4 == 1 && odgk3 == 1) tacni++;
            if (p4 == 1 && odgk3 == 0) greska = true;

            if (p5 == 1 && odgk4 == 1) tacni++;
            if (p5 == 1 && odgk4 == 0) greska = true;

            int osvojeniPoeni = 0;

            if (greska)
            {
                osvojeniPoeni = -10;
            }

            if (!greska)
       
            {
                if (tacni == 3)
                    osvojeniPoeni = 30;
                else if (tacni == 2)
                    osvojeniPoeni = 20;
                else if (tacni == 1)
                    osvojeniPoeni = 10;

               
            }

            poeni += osvojeniPoeni;

            label12.Text = "Poeni: " + poeni;

            if (progressBar1.Value < progressBar1.Maximum)
                progressBar1.Value += 1;
            else
            {
                MessageBox.Show("Kraj kviza sa osvojenih poena:" + poeni);
                Reset();
                timer1.Stop();
                return;
            }
            checkBox1.Checked = false;
            checkBox2.Checked = false;
            checkBox3.Checked = false;
            checkBox4.Checked = false;
            checkBox5.Checked = false;
            button2.PerformClick();
        }
        private void Reset()
        {
            timer1.Start();
            poeni = 0;
            label12.Text = "Poeni:" + poeni;
            preskoceno = 0;
            label6.Text = "Preskocena:" + preskoceno;
            time = 0;
            label5.Text = "Vreme:" + preskoceno;
            progressBar1.Value = 0;
        }
