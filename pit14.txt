        private void UcitajPodatke(string putanja, CheckedListBox clb, ListBox lb)
        {
            var linije = File.ReadAllLines(putanja);

            foreach (var linija in linije)
            {
                string[] delovi = linija.Split(';');
                clb.Items.Add(delovi[0]);
                lb.Items.Add(delovi[1]);
            }
        }
