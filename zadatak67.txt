using System;

    public partial class Form1 : Form
    {
        string putanja = "snimi.txt";
        public Form1()
        {
            InitializeComponent();
            button6.Left += 1000;
            foreach (string linija in File.ReadLines("tekst.txt"))
            {
                richTextBox1.AppendText(linija + Environment.NewLine);
            }
        }

        private void button1_Click(object sender, EventArgs e)
        {
            string ulaz = richTextBox1.Text;
            string sifrovano = "";

            foreach (char c in ulaz)
            {
                int ascii = (int)c;
                ascii += 3;
                sifrovano += (char)ascii;
            }


            richTextBox2.Text = sifrovano;
        }

        private void button2_Click(object sender, EventArgs e)
        {
            string putanja = "snimi.txt";
            File.WriteAllText(putanja, richTextBox2.Text);
            button6.PerformClick();
        }

        private void button3_Click(object sender, EventArgs e)
        {
            foreach (string linija in File.ReadLines("snimi.txt"))
            {
                richTextBox1.AppendText(linija + Environment.NewLine);
            }
        }

        private void button6_Click(object sender, EventArgs e)
        {
            richTextBox1.Text = "";
            richTextBox2.Text = "";
        }

        private void button4_Click(object sender, EventArgs e)
        {
            string ulaz = richTextBox1.Text;
            string sifrovano = "";

            foreach (char c in ulaz)
            {
                int ascii = (int)c;
                ascii -= 3;
                sifrovano += (char)ascii;
            }


            richTextBox2.Text = sifrovano;
        }

        private void button5_Click(object sender, EventArgs e)
        {
            string ulaz = richTextBox1.Text;
            StringBuilder sb = new StringBuilder();

            for (int i = 0; i < ulaz.Length; i++)
            {
                char c = ulaz[i];

                if (i % 2 == 0)
                {
                    sb.Append((char)(c + 3));
                }
                else
                {
                    sb.Append((char)(c + 10));
                }
            }

            richTextBox2.Text = sb.ToString();
        }

        private void button7_Click(object sender, EventArgs e)
        {
            string ulaz = richTextBox1.Text;
            StringBuilder sb = new StringBuilder();

            for (int i = 0; i < ulaz.Length; i++)
            {
                char c = ulaz[i];

                if (i % 2 == 0)
                {
                    sb.Append((char)(c - 3));
                }
                else
                {
                    sb.Append((char)(c - 10));
                }

            }

            richTextBox2.Text = sb.ToString();
        }

        private void button8_Click(object sender, EventArgs e)
        {
            string ulaz = richTextBox1.Text;
            StringBuilder sb = new StringBuilder();
            foreach (char c in ulaz)
            {
                switch (c)
                {
                    case 'a':
                        sb.Append('@');
                        break;

                    case 'b':
                        sb.Append('8');
                        break;

                    case 'c':
                        sb.Append('(');
                        break;

                    case 'd':
                        sb.Append('D');
                        break;

                    case 'e':
                        sb.Append('3');
                        break;

                    case 'f':
                        sb.Append("p");
                        break;

                    case 'g':
                        sb.Append('9');
                        break;

                    case 'h':
                        sb.Append("#");
                        break;

                    case 'i':
                        sb.Append('!');
                        break;

                    case 'j':
                        sb.Append("j7");
                        break;

                    default:

                        sb.Append((char)(c + 6));
                        break;
                }
            }

            richTextBox2.Text = sb.ToString();
        }

        private void button9_Click(object sender, EventArgs e)
        {
            string ulaz = richTextBox1.Text;
            StringBuilder sb = new StringBuilder();
            foreach (char c in ulaz)
            {
                switch (c)
                {
                    case '@':
                        sb.Append('a');
                        break;

                    case '8':
                        sb.Append('b');
                        break;

                    case '(':
                        sb.Append('c');
                        break;

                    case 'd':
                        sb.Append('D');
                        break;

                    case '3':
                        sb.Append('e');
                        break;

                    case 'p':
                        sb.Append("f");
                        break;

                    case '9':
                        sb.Append('g');
                        break;

                    case '#':
                        sb.Append("h");
                        break;

                    case '!':
                        sb.Append('i');
                        break;

                    case '7':
                        sb.Append("j");
                        break;

                    default:

                        sb.Append((char)(c + 6));
                        break;
                }
            }

            richTextBox2.Text = sb.ToString();
        }
