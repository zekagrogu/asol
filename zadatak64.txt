        private void KlikPolja(int index, Button btn)
        {
            if (igraGotova) return;

            if (tabla[index] == '\0')
            {
                tabla[index] = igrac;
                btn.Text = igrac.ToString();
                btn.Enabled = false;

                if (ProveriPobedu())
                {
                    MessageBox.Show("Pobednik: " + igrac);
                   
                    tabControl1.SelectedIndex = 0;
                    return;
                }
                potezi.Push(index);

                igrac = (igrac == 'X') ? 'O' : 'X';
            }
        }

        private bool ProveriPobedu()
        {
            int[,] win = {
        {0,1,2},{3,4,5},{6,7,8},
        {0,3,6},{1,4,7},{2,5,8},
        {0,4,8},{2,4,6}
    };

            for (int i = 0; i < win.GetLength(0); i++)
            {
                if (tabla[win[i, 0]] != '\0' &&
                    tabla[win[i, 0]] == tabla[win[i, 1]] &&
                    tabla[win[i, 1]] == tabla[win[i, 2]])
                {
                    return true;
                }
            }

            return false;
        }

        private void KreirajTablu8x8()
        {
            int size = 45; // veća dugmad

            dugmad = new Button[8, 8];

                    // 💜
                    dugmad[i, j].BackColor = Color.LightYellow;
                    dugmad[i, j].ForeColor = Color.Blue;
                   
                    dugmad[i, j].FlatStyle = FlatStyle.Flat;
                    dugmad[i, j].FlatAppearance.BorderColor = Color.Blue;
                    dugmad[i, j].FlatAppearance.BorderSize = 1;

        private void Klik8(int i, int j, Button btn)
            {
                if (igraGotova) return;

                int kolona = j;

                // tražimo odozdo prvo slobodno mesto
                for (int red = 7; red >= 0; red--)
                {
                    if (tabla8[red, kolona] == '\0')
                    {
                        tabla8[red, kolona] = igrac;
                        dugmad[red, kolona].Text = igrac.ToString();
                        dugmad[red, kolona].Enabled = false;

                        if (Proveri4(red, kolona))
                        {
                            MessageBox.Show("Pobednik: " + igrac);
                         
                            tabControl1.SelectedIndex = 0;
                            return;
                        }

                        igrac = (igrac == 'X') ? 'O' : 'X';
                   
                    return;
                    }
                }

                MessageBox.Show("Kolona je puna!");
            }

           


        private bool Proveri4(int x, int y)
        {
            return ProveriPravac(x, y, 1, 0) || // horizontalno
                   ProveriPravac(x, y, 0, 1) || // vertikalno
                   ProveriPravac(x, y, 1, 1) || // dijagonala
                   ProveriPravac(x, y, 1, -1);  // druga dijagonala
        }

        private bool ProveriPravac(int x, int y, int dx, int dy)
        {
            char p = tabla8[x, y];
            int count = 1;

            for (int i = 1; i < 4; i++)
            {
                int nx = x + dx * i;
                int ny = y + dy * i;

                if (nx >= 0 && nx < 8 && ny >= 0 && ny < 8 &&
                    tabla8[nx, ny] == p)
                    count++;
                else break;
            }

            for (int i = 1; i < 4; i++)
            {
                int nx = x - dx * i;
                int ny = y - dy * i;

                if (nx >= 0 && nx < 8 && ny >= 0 && ny < 8 &&
                    tabla8[nx, ny] == p)
                    count++;
                else break;
            }

            return count >= 4;
        }
