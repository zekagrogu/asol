using System;
using System.Collections.Generic;
using System.ComponentModel;
using System.Data;
using System.Drawing;
using System.Linq;
using System.Text;
using System.Threading.Tasks;
using System.Windows.Forms;

namespace labele
{
    public partial class Form1 : Form
    {
        int pom = 25;
        int pom2 = 10;
        int py = 1;
        int px = 1;
        public Form1()
        {
            InitializeComponent();
            tabControl1.Appearance = TabAppearance.FlatButtons;
            tabControl1.ItemSize = new Size(0, 1);
            tabControl1.SizeMode = TabSizeMode.Fixed;

        }

        private void button1_Click(object sender, EventArgs e)
        {
            label2.Text = textBox1.Text;
            this.ControlBox = false;
            pictureBox1.Dock = DockStyle.None;
            this.FormBorderStyle = FormBorderStyle.None;
            tabControl1.SelectedIndex = 1;
            tabControl1.TabPages[1].BackColor = Color.Black;
            this.WindowState = FormWindowState.Maximized;
            label2.Left = this.Width;
            timer1.Start();
            pictureBox1.Left = 5000;

        }

        private void button2_Click(object sender, EventArgs e)
        {
            label2.Text = textBox1.Text;
            pictureBox1.Dock = DockStyle.None;
            this.ControlBox = false;
            this.FormBorderStyle = FormBorderStyle.None;
            tabControl1.SelectedIndex = 1;
            tabControl1.TabPages[1].BackColor = Color.Black;
            this.WindowState = FormWindowState.Maximized;
            label2.Left = this.Width/2;
            label2.Top = this.Height / 2;
            timer2.Start();
            pictureBox1.Left = 5000;
        }

        private void button3_Click(object sender, EventArgs e)
        {
            label2.Text = textBox1.Text;
            this.ControlBox = false;
            pictureBox1.Dock = DockStyle.None;
            this.FormBorderStyle = FormBorderStyle.None;
            tabControl1.SelectedIndex = 1;
            tabControl1.TabPages[1].BackColor = Color.Black;
            this.WindowState = FormWindowState.Maximized;
            label2.Left = this.Width;
            timer3.Start();
            label2.Left = 5000;
        }

        private void timer1_Tick(object sender, EventArgs e)
        {
            label2.Left -= pom;
            if(label2.Left<(-1)*label2.Width)
            {
                label2.Left = this.Width;
                Random x = new Random();
                int poz = x.Next(0, this.Height = label2.Height);
                label2.Top = poz;
            }
        }

        private void tabControl1_KeyPress(object sender, KeyPressEventArgs e)
        {
            if (e.KeyChar == 's')
            {
                tabControl1.SelectedIndex = 0;
                this.ControlBox = true;
                this.FormBorderStyle = FormBorderStyle.Sizable;
                this.WindowState = FormWindowState.Normal;
                timer1.Stop();

            }
        }

        private void timer2_Tick(object sender, EventArgs e)
        {
            Random x = new Random();
            int a = x.Next(0, 255);
            int b = x.Next(0, 255);
            int c = x.Next(0, 255);
            if ((px == 1) && (py == 1)) { label2.Left += pom2; label2.Top += pom2; }
            if ((px == 1) && (py == -1)) { label2.Left += pom2; label2.Top -= pom2; }
            if ((px == -1) && (py == -1)) { label2.Left -= pom2; label2.Top -= pom2; }
            if ((px == -1) && (py == 1)) { label2.Left -= pom2; label2.Top += pom2; }

            if (label2.Top + label2.Height  > this.Height)
            {
                py *= (-1);
                label2.ForeColor = Color.FromArgb(255, a, b, c);
            }

            if (label2.Top < 0)
            {
                py *= (-1);
                label2.ForeColor = Color.FromArgb(255, a, b, c);
            }

            if (label2.Left + label2.Width > this.Width)
            {
                px *= (-1);
                label2.ForeColor = Color.FromArgb(255, a, b, c);
            }

            if (label2.Left < 0)
            {
                px *= (-1);
                label2.ForeColor = Color.FromArgb(255, a, b, c);
            }


        }
    }
}
